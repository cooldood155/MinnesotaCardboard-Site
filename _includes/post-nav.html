{%- comment -%} "Older" and "Newer" links within a list of posts. Left out when the post has no neighbors in the list. assets/js/post-nav.js combines lists with the same links and lets readers switch between the rest. site.posts is newest first. Example use: {% include post-nav.html posts=live_posts list="all" label="All posts" %} {%- endcomment -%} {%- assign nav_older = nil -%} {%- assign nav_newer = nil -%} {%- for nav_post in include.posts -%} {%- if nav_post.url == page.url -%} {%- assign nav_older = include.posts[forloop.index] -%} {%- unless forloop.first -%} {%- assign nav_newer_index = forloop.index0 | minus: 1 -%} {%- assign nav_newer = include.posts[nav_newer_index] -%} {%- endunless -%} {%- endif -%} {%- endfor -%} {%- if nav_older or nav_newer %}
{%- if nav_older %} ← Older {{ nav_older.title | escape }} {%- endif %} {%- if nav_newer %} Newer → {{ nav_newer.title | escape }} {%- endif %}
{%- endif -%}
